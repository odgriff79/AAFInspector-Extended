import os
import json
import urllib.parse
import tkinter as tk
from tkinter import filedialog, scrolledtext, messagebox

# --- Core logic from your provided script ---

def create_mob_map(node, mob_map=None):
    """Recursively builds a dictionary mapping MobIDs to their corresponding nodes."""
    if mob_map is None:
        mob_map = {}
    if not isinstance(node, list) or len(node) < 2:
        return mob_map
    
    children = node[3] if len(node) > 3 else []
    
    # If the node has a MobID, add it to the map
    if any(isinstance(c, list) and c[0] == "MobID" for c in children):
        mobid = next((c[2] for c in children if isinstance(c, list) and c[0] == "MobID"), None)
        if mobid:
            mob_map[mobid] = node
            
    # Recurse through all children
    for c in children:
        create_mob_map(c, mob_map)
        
    return mob_map

def extract_metadata(mob_node):
    """Extracts the URLString from a mob node."""
    metadata = {"URLString": ""}
    
    def recurse(n):
        if not isinstance(n, list):
            return
            
        name = n[0] if len(n) >= 1 else ""
        children = n[3] if len(n) > 3 else []
        
        # Find the URLString property
        if name == "URLString" and len(n) > 2:
            metadata["URLString"] = n[2]
            
        # Recurse through children
        for c in children:
            recurse(c)
            
    recurse(mob_node)
    return metadata

def find_clips_in_sequence(node, results=None, dedupe_set=None):
    """
    Recursively searches for SourceClips within a Sequence and extracts
    their MobID and StartTime.
    """
    if results is None: results = []
    if dedupe_set is None: dedupe_set = set()
    if not isinstance(node, list) or len(node) < 2: return results

    name = node[0]
    children = node[3] if len(node) > 3 else []

    # If the node is a Sequence, traverse its Components
    if name == "Sequence":
        components_node = next((c for c in children if isinstance(c, list) and c[0] == "Components"), None)
        if components_node and len(components_node) > 3:
            for component in components_node[3]:
                find_clips_in_sequence(component, results, dedupe_set)

    # If the node is a SourceClip, extract its data
    elif name == "SourceClip":
        mobid = ""
        start_time = 0
        for c in children:
            if isinstance(c, list):
                if c[0] == "SourceID":
                    mobid = c[2]
                elif c[0] == "StartTime":
                    try:
                        start_time = int(c[2])
                    except (ValueError, TypeError):
                        start_time = 0
        
        # Add the clip if it has a MobID and is unique
        dedupe_key = (mobid, start_time)
        if mobid and dedupe_key not in dedupe_set:
            dedupe_set.add(dedupe_key)
            results.append({"MobID": mobid, "StartTime": start_time})
    
    # For other node types, continue the search through their children
    else:
        for child in children:
            find_clips_in_sequence(child, results, dedupe_set)
            
    return results

# --- Simple GUI Application ---

class SequenceViewerApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Sequence Source File Viewer")
        self.root.geometry("900x600")

        self.json_data = None

        # --- Widgets ---
        top_frame = tk.Frame(root)
        top_frame.pack(pady=10, padx=10, fill=tk.X)

        tk.Button(top_frame, text="Load Sequence JSON File", command=self.load_json).pack(side=tk.LEFT)
        tk.Button(top_frame, text="Find Source Files", command=self.process_json).pack(side=tk.LEFT, padx=10)
        
        self.results_text = scrolledtext.ScrolledText(root, wrap=tk.WORD, font=("Courier New", 10))
        self.results_text.pack(pady=5, padx=10, expand=True, fill=tk.BOTH)

    def log(self, message):
        """Adds a message to the results text area."""
        self.results_text.insert(tk.END, message + "\n")
        self.results_text.see(tk.END)

    def load_json(self):
        """Opens a file dialog to load the compressed JSON file."""
        path = filedialog.askopenfilename(
            title="Select Compressed JSON",
            filetypes=[("JSON Files", "*.json")]
        )
        if not path:
            return
        
        try:
            with open(path, "r", encoding="utf-8") as f:
                self.json_data = json.load(f)
            self.results_text.delete(1.0, tk.END)
            self.log(f"✅ Successfully loaded: {os.path.basename(path)}")
        except Exception as e:
            messagebox.showerror("Error", f"Failed to load or parse JSON file.\n\n{e}")
            self.json_data = None

    def process_json(self):
        """Processes the loaded JSON to find and display source files."""
        if not self.json_data:
            messagebox.showwarning("No Data", "Please load a JSON file first.")
            return

        self.results_text.delete(1.0, tk.END)
        self.log("🔎 Starting analysis...")

        try:
            # 1. Build a map of all MobIDs for quick lookup
            self.log("    - Indexing all media objects (Mobs)...")
            mob_map = create_mob_map(self.json_data)
            self.log(f"    - Found {len(mob_map)} unique Mobs.")

            # 2. Find all source clips within the main sequence
            self.log("    - Searching for source clips in the sequence...")
            clips = find_clips_in_sequence(self.json_data)
            if not clips:
                self.log("\n❌ No source clips were found in any sequence.")
                messagebox.showinfo("Done", "Could not find any source clips.")
                return

            self.log(f"    - Found {len(clips)} clips in the sequence.\n")
            self.log(f"{'Source File':<60}{'Start Time'}")
            self.log(f"{'-'*60}{'-'*12}")

            # 3. Get metadata for each clip and display it
            for clip in clips:
                mob_id = clip["MobID"]
                start_time = clip["StartTime"]
                mob_node = mob_map.get(mob_id)
                
                source_file = "Filename not found"
                if mob_node:
                    metadata = extract_metadata(mob_node)
                    url_string = metadata.get("URLString")
                    if url_string:
                        # Decode URL encoding (e.g., %20 -> space) and get the filename
                        decoded_path = urllib.parse.unquote(urllib.parse.urlparse(url_string).path)
                        source_file = os.path.basename(decoded_path)

                self.log(f"{source_file:<60}{start_time}")
            
            self.log("\n✅ Analysis complete.")
            messagebox.showinfo("Done", "Finished processing the sequence.")

        except Exception as e:
            self.log(f"\n❌ An error occurred: {e}")
            messagebox.showerror("Processing Error", f"An error occurred during analysis.\n\n{e}")

# --- Main execution ---
if __name__ == "__main__":
    root = tk.Tk()
    app = SequenceViewerApp(root)
    root.mainloop()
